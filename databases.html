Database enhancement and portfolio value

The enhanced artifact is AURELIA Wealth Studio, a Windows desktop application built in Visual Studio 2022 with C++17 and WinSQLite. The relational database connects clients with accounts, holdings, securities, transactions, portfolio snapshots, financial goals, advisor tasks, notes and audit events. Separate tables make relationships explicit and reduce repeated information. Foreign keys and integrity checks help identify broken relationships.

I selected this artifact because it demonstrates how an existing system can become more reliable while retaining its earlier functionality. The implementation uses prepared statements to bind values, transactions to protect related changes, validation before saving and additive migrations that preserve existing records. Client profiles, financial goal forms and the advisor ticket desk connect database operations to practical application workflows.

The demonstration contains 25 distinct fictional clients. Dashboard calculations and reports use stored holdings and modeled security prices. Individual PDF reports and Excel workbooks make those records reviewable, and native Excel charts use worksheet data. These components demonstrate that persistence supports both client management and communication.

Learning and challenges

The most useful lesson from this enhancement is that saving data successfully does not establish data integrity. Related changes need clear rules, and those rules must be tested when an operation fails. Prepared statements, transactions and foreign keys connected database design with the defensive programming approach established in my earlier enhancement.

Preserving the demonstration records was an important challenge. Duplicate profiles were converted into distinct fictional identities after a backup while retaining their identifiers and related financial records. Startup seeding also needed to avoid creating those identities again. Comparing the client directory, dashboard and exported reports with SQLite calculations helped identify discrepancies that visual inspection alone could miss.

Testing exposed a Windows SQLite date issue affecting overdue tasks. Correcting the business date calculation restored the expected six overdue tickets. The final Debug x64 solution build completed with zero errors and 24 nonfatal conversion warnings. Database workflow tests passed 111 assertions on an isolated copy, and ticket and client interface tests passed 298 assertions. All 26 PDFs, 26 workbooks and 52 native Excel charts were checked.

static std::wstring widen(
    const std::string& value) {

    if (value.empty()) {

        return L"";
    }

    const int size =
        MultiByteToWideChar(
            CP_UTF8,
            0,
            value.c_str(),
            static_cast<int>(
                value.size()),
            nullptr,
            0);

    std::wstring result(
        size,
        L'\0');

    MultiByteToWideChar(
        CP_UTF8,
        0,
        value.c_str(),
        static_cast<int>(
            value.size()),
        result.data(),
        size);

    return result;
}

static HFONT createFont(
    HWND reference,
    int points,
    bool bold =
    false) {

    HDC deviceContext =
        GetDC(
            reference);

    const int dpi =
        GetDeviceCaps(
            deviceContext,
            LOGPIXELSY);

    ReleaseDC(
        reference,
        deviceContext);

    LOGFONTW fontData = {};

    fontData.lfHeight =
        -MulDiv(
            points,
            dpi,
            72);

    fontData.lfWeight =
        bold
        ? FW_SEMIBOLD
        : FW_NORMAL;

    wcscpy_s(
        fontData.lfFaceName,
        L"Segoe UI");

    return CreateFontIndirectW(
        &fontData);
}

static void drawRoundedBox(
    Graphics& graphics,
    REAL x,
    REAL y,
    REAL width,
    REAL height,
    REAL radius,
    const Color& fill,
    const Color& border =
    Color(
        0,
        0,
        0,
        0)) {

    GraphicsPath path;

    const REAL diameter =
        radius *
        2.0f;

    path.AddArc(
        x,
        y,
        diameter,
        diameter,
        180,
        90);

    path.AddArc(
        x +
        width -
        diameter,
        y,
        diameter,
        diameter,
        270,
        90);

    path.AddArc(
        x +
        width -
        diameter,
        y +
        height -
        diameter,
        diameter,
        diameter,
        0,
        90);

    path.AddArc(
        x,
        y +
        height -
        diameter,
        diameter,
        diameter,
        90,
        90);

    path.CloseFigure();

    SolidBrush fillBrush(
        fill);

    graphics.FillPath(
        &fillBrush,
        &path);

    if (
        border.GetA() >
        0) {

        Pen borderPen(
            border,
            1.0f);

        graphics.DrawPath(
            &borderPen,
            &path);
    }
}

            controlText(

            configuredUsername =
            environmentValue(

        if (

            showMessage(

                L"CS499_USERNAME=demo_admin\n"
                L"CS499_PASSWORD=AureliaDemo2026");

            return;
        }


        if (
            username !=
            configuredUsername ||
            password !=
            configuredPassword) {

            failedLoginAttempts++;


            logger.write(
                "GUI_AUTH_FAILURE");


            showMessage(
                window,
                L"Username or password "
                L"was not recognized.");


            if (
                failedLoginAttempts >=
                3) {

                EnableWindow(
                    usernameField,
                    FALSE);

                EnableWindow(
                    passwordField,
                    FALSE);
            }


            return;
        }


        authenticated =
            true;


        failedLoginAttempts =
            0;


        SetWindowTextW(
            passwordField,
            L"");


        logger.write(
            "GUI_AUTH_SUCCESS");


        navigate(
            Page::Dashboard);
    }

    void cleanLegacyDemoDuplicates() {
        int number = demoDuplicateCount();
        if (!number) {
            showMessage(window, L"No known legacy fictional duplicates found.");
            return;
        }
        std::wstring question =
            L"Found " + std::to_wstring(number) +
            L" duplicate fictional investor records.\n\n"
            L"Create a SQLite backup and remove ONLY the "
            L"verified 'demo.investor' duplicates?\n"
            L"Do not use this feature for real financial data.";
        if (MessageBoxW(window, question.c_str(),
            L"Back up and clean demo clients",
            MB_YESNO | MB_ICONWARNING) != IDYES) return;

        const std::string backupName =
            database.fileName() + ".before_demo_cleanup.bak";
        sqlite3* backupDb = nullptr;
        if (sqlite3_open(backupName.c_str(), &backupDb) != SQLITE_OK) {
            if (backupDb) sqlite3_close(backupDb);
            throw DatabaseException("Unable to open backup file");
        }
        sqlite3_backup* copy = sqlite3_backup_init(
            backupDb, "main", database.handle(), "main");
        if (!copy) {
            string reason = sqlite3_errmsg(backupDb);
            sqlite3_close(backupDb);
            throw DatabaseException("Backup initialization failed: " + reason);
        }
        const int rc = sqlite3_backup_step(copy, -1);
        const int finished = sqlite3_backup_finish(copy);
        sqlite3_close(backupDb);
        if (rc != SQLITE_DONE || finished != SQLITE_OK)
            throw DatabaseException("Backup incomplete. Nothing deleted.");

        Transaction tx(database);
        database.execute(
            "DELETE FROM clients WHERE id IN ("
            "SELECT d.id FROM clients d JOIN clients p "
            "ON d.id<>p.id "
            "AND lower(d.name)=lower(p.name) "
            "AND d.age=p.age "
            "AND abs(d.account_balance-p.account_balance)<0.01 "
            "WHERE d.email LIKE 'demo.investor%@aurelia.example' "
            "AND p.email LIKE '%@example.com' "
            "AND p.email NOT LIKE 'demo.investor%@aurelia.example');");
        const int deleted = database.changes();
        repository.recordAuditEvent(
            "DEMO_DUPLICATES_CLEANED", 0,
            std::to_string(deleted) + " exact demo-only records consolidated; backup: " + backupName);
        tx.commit();
        showMessage(window,
            L"Consolidated " + std::to_wstring(deleted) +
            L" verified demo records.\nA backup was saved alongside the SQLite file.");
        updateLayout();
    }

static LRESULT CALLBACK MainWindowProc(
    HWND window,
    UINT message,
    WPARAM wParam,
    LPARAM lParam) {

    AureliaApplication* app =
        reinterpret_cast<
        AureliaApplication*>(
            GetWindowLongPtrW(
                window,
                GWLP_USERDATA));


    if (
        message ==
        WM_NCCREATE) {

        CREATESTRUCTW* create =
            reinterpret_cast<
            CREATESTRUCTW*>(
                lParam);


        app =
            reinterpret_cast<
            AureliaApplication*>(
                create
                ->lpCreateParams);


        SetWindowLongPtrW(
            window,
            GWLP_USERDATA,
            reinterpret_cast<
            LONG_PTR>(
                app));


        return TRUE;
    }


    if (!app) {

        return DefWindowProcW(
            window,
            message,
            wParam,
            lParam);
    }


    switch (
        message) {
    case WM_TIMER:
        if(wParam==28){app->pollMarketObservation();return 0;}
        if (wParam == 27 && app->currentPage == Page::Loading) {
            ++app->loadingFrame;
            InvalidateRect(window, nullptr, FALSE);
            if (app->loadingFrame >= 3) {
                KillTimer(window, 27);
                app->currentPage = app->loadingDestination;
                app->updateLayout();
            }
            return 0;
        }
        break;


    case WM_SIZE:

        app->updateLayout();

        return 0;


    case WM_COMMAND:

        if (
            LOWORD(
                wParam) ==
            1003 &&
            HIWORD(
                wParam) ==
            EN_CHANGE &&
            app->currentPage ==
            Page::Clients) {

            app->refreshClientList();
        }

        return 0;


    case WM_NOTIFY:
    {
        NMHDR* header =
            reinterpret_cast<
            NMHDR*>(
                lParam);


        if (
            header &&
            header->idFrom ==
            1004 &&
            header->code ==
            NM_DBLCLK &&
            app->currentPage ==
            Page::Clients) {

            app->handleAction(
                ACTION_OPEN_CLIENT);

            return 0;
        }


        if (header && app->isTaskList(header->hwndFrom) && header->code == LVN_GETINFOTIP) {
            auto* tip = reinterpret_cast<NMLVGETINFOTIPW*>(lParam);
            int index = app->taskRowIndex(header->hwndFrom, tip->iItem);
            if (index >= 0 && index < static_cast<int>(app->displayedTasks.size())) {
                const auto& t = app->displayedTasks[index];
                auto client = t.clientId ? app->repository.findClientById(*t.clientId) : std::optional<Client>{};
                std::wstring text = widen("AW-" + std::to_string(t.id) + " | " + (client ? client->name : "Firm wide") + "\n" + t.title + "\nPriority: " + t.priority + " | Status: " + t.status + "\nDue: " + t.dueDate + " | Assigned to: " + t.assignedTo + "\nCategory: " + t.category + "\n" + t.description);
                wcsncpy_s(tip->pszText, tip->cchTextMax, text.c_str(), _TRUNCATE);
            }
            return 0;
        }
        if (header && header->code == LVN_COLUMNCLICK) {
            int column = reinterpret_cast<NMLISTVIEW*>(lParam)->iSubItem;
            if (app->isTaskList(header->hwndFrom)) {
                app->taskSortAscending = app->taskSortColumn == column ? !app->taskSortAscending : true;
                app->taskSortColumn = column; app->refreshTasksList(); return 0;
            }
            if (header->hwndFrom == app->clientList) {
                app->clientSortAscending = app->clientSortColumn == column ? !app->clientSortAscending : true;
                app->clientSortColumn = column; app->refreshClientList(); return 0;
            }
        }
        if (header && app->isTaskList(header->hwndFrom) && header->code == LVN_ITEMCHANGED) {
            auto* change = reinterpret_cast<NMLISTVIEW*>(lParam);
            if ((change->uNewState & LVIS_SELECTED) && !(change->uOldState & LVIS_SELECTED)) {
                for (HWND lane : { app->tasksList, app->mediumTasksList, app->completedTasksList }) if (lane != header->hwndFrom) ListView_SetItemState(lane, -1, 0, LVIS_SELECTED);
                InvalidateRect(app->window, nullptr, FALSE);
            }
        }
        if(header && app->isTaskList(header->hwndFrom) && header->code==NM_CUSTOMDRAW){
            auto* cd=reinterpret_cast<NMLVCUSTOMDRAW*>(lParam);
            if(cd->nmcd.dwDrawStage==CDDS_PREPAINT)return CDRF_NOTIFYITEMDRAW;
            if(cd->nmcd.dwDrawStage==CDDS_ITEMPREPAINT){size_t row=static_cast<size_t>(app->taskRowIndex(header->hwndFrom, static_cast<int>(cd->nmcd.dwItemSpec)));if(row<app->displayedTasks.size()){const auto& t=app->displayedTasks[row];cd->clrText=t.completed?RGB(110,110,110):(t.priority=="High"||t.priority=="Urgent"?RGB(155,65,55):RGB(29,38,42));cd->clrTextBk=row%2?RGB(249,246,240):RGB(255,255,255);}return CDRF_DODEFAULT;}
        }
        if (header && header->code == NM_DBLCLK &&
            app->isTaskList(header->hwndFrom) &&
            app->currentPage == Page::AdvisorTasks) {
            app->handleAction(ACTION_TASK_CLIENT);
            return 0;
        }
        if (header && header->code == NM_DBLCLK &&
            header->idFrom == 1007 &&
            app->currentPage == Page::FinancialGoals) {
            app->handleAction(ACTION_GOAL_CLIENT);
            return 0;
        }

        break;
    }


    case WM_LBUTTONUP:
    {
        POINT point = {

            static_cast<short>(
                LOWORD(
                    lParam)),

            static_cast<short>(
                HIWORD(
                    lParam))
        };


        for (
            const ClickArea& area :
            app->clickAreas) {

            if (
                PtInRect(
                    &area.rectangle,
                    point)) {

                app->handleAction(
                    area.action);

                return 0;
            }
        }


        return 0;
    }


    case WM_SETCURSOR:
    {
        if (
            LOWORD(
                lParam) ==
            HTCLIENT) {

            POINT cursor = {};


            GetCursorPos(
                &cursor);


            ScreenToClient(
                window,
                &cursor);


            for (
                const ClickArea& area :
                app->clickAreas) {

                if (
                    PtInRect(
                        &area.rectangle,
                        cursor)) {

                    SetCursor(
                        LoadCursorW(
                            nullptr,
                            IDC_HAND));

                    return TRUE;
                }
            }
        }


        break;
    }


    case WM_PAINT:
    {
        PAINTSTRUCT paint = {};


        HDC deviceContext =
            BeginPaint(
                window,
                &paint);


        {
            Graphics graphics(
                deviceContext);


            try {

                app->paint(
                    graphics);
            }


            catch (
                const std::exception& exception) {

                OutputDebugStringA(
                    exception.what());
            }
        }


        EndPaint(
            window,
            &paint);


        return 0;
    }


    case WM_DESTROY:

        PostQuitMessage(
            0);

        return 0;


    default:

        break;
    }


    return DefWindowProcW(
        window,
        message,
        wParam,
        lParam);
}

int WINAPI wWinMain(
    HINSTANCE instance,
    HINSTANCE,
    LPWSTR,
    int) {

    INITCOMMONCONTROLSEX controls = {};


    controls.dwSize =
        sizeof(
            controls);


    controls.dwICC =
        ICC_LISTVIEW_CLASSES |
        ICC_STANDARD_CLASSES;


    InitCommonControlsEx(
        &controls);


    GdiplusStartupInput startupInput;


    ULONG_PTR gdiplusToken =
        0;


    if (
        GdiplusStartup(
            &gdiplusToken,
            &startupInput,
            nullptr) !=
        Ok) {

        MessageBoxW(
            nullptr,
            L"Unable to initialize graphical interface.",
            L"Aurelia Wealth Studio Startup Error",
            MB_OK |
            MB_ICONERROR);

        return 1;
    }


    int exitCode =
        0;


    try {

        AureliaApplication app(
            instance);


        app.createMainWindow();


        MSG message = {};


        while (
            GetMessageW(
                &message,
                nullptr,
                0,
                0) >
            0) {

            TranslateMessage(
                &message);


            DispatchMessageW(
                &message);
        }


        exitCode =
            static_cast<int>(
                message.wParam);
    }


    catch (
        const std::exception& exception) {

        MessageBoxW(
            nullptr,
            widen(
                exception.what())
            .c_str(),
            L"Aurelia Wealth Studio Startup Error",
            MB_OK |
            MB_ICONERROR);


        exitCode =
            1;
    }


    GdiplusShutdown(
        gdiplusToken);


    return exitCode;
}
